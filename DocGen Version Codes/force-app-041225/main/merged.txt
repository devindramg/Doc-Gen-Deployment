* @description   Manages AI model configurations and orchestrates AI processing requests.

public with sharing class AIModelManager {
    
    //======================================================
    // DTO WRAPPER CLASS FOR AI MODELS
    //======================================================
    public class AIModelConfig {
        
        @AuraEnabled public String modelName        { get; set; }
        @AuraEnabled public String modelIdentifier  { get; set; }
        @AuraEnabled public String provider         { get; set; }
        @AuraEnabled public String capabilities     { get; set; }
        @AuraEnabled public Boolean isActive        { get; set; }   // Added for ClauseGenerator checks
        
        // Construct from Custom Metadata record
        public AIModelConfig(AI_Model_Configuration__mdt config) {
            this.modelName       = config.ModelName__c;
            this.modelIdentifier = config.Model_Id__c;
            this.provider        = config.Provider__c;
            this.capabilities    = config.Capabilities__c;
            this.isActive        = config.Is_Active__c;
        }
        
        // Construct for hardcoded native Einstein models
        public AIModelConfig(String name, String identifier, String provider, String capabilities) {
            this.modelName       = name;
            this.modelIdentifier = identifier;
            this.provider        = provider;
            this.capabilities    = capabilities;
            this.isActive        = true;     // Native models are always active
        }
    }
    
    //======================================================
    // GET AVAILABLE AI MODELS (for UI dropdown)
    //======================================================
    @AuraEnabled(cacheable=true)
    public static List<AIModelConfig> getAvailableAIModels() {
        List<AIModelConfig> models = new List<AIModelConfig>();
        
        try {
            // From Custom Metadata
            for (AI_Model_Configuration__mdt config :
                 [SELECT ModelName__c, Model_Id__c, Provider__c, Capabilities__c, Is_Active__c
                  FROM AI_Model_Configuration__mdt
                  WHERE Is_Active__c = true
                  ORDER BY ModelName__c]
                ) {
                    models.add(new AIModelConfig(config));
                }
            
            // Add hardcoded Einstein models
            models.add(new AIModelConfig('Einstein Intent Detection',
                                         'einstein_intent',
                                         'Salesforce',
                                         'Intent Analysis, Classification'));
            
            models.add(new AIModelConfig('Einstein Sentiment Analysis',

            models.add(new AIModelConfig('Einstein OCR',
                                         'einstein_ocr',
                                         'Salesforce',
                                         'Text Extraction, Document Parsing'));
            
        } catch (Exception e) {
            System.debug(LoggingLevel.ERROR,
                         'Error retrieving AI models: ' + e.getMessage());
            throw new AuraHandledException('Failed to retrieve AI models: ' + e.getMessage());
        }
        
        return models;

    /**
* @description   Finds an AI model by its unique identifier.
* @param         modelIdentifier
* @return        AIModelConfig or null
*/
    public static AIModelConfig getAIModelByIdentifier(String modelIdentifier) {
        if (String.isBlank(modelIdentifier)) {
            return null;
        }
        
        for (AIModelConfig model : getAvailableAIModels()) {
            if (model.modelIdentifier == modelIdentifier) {
                return model;
            }
        }
        
        return null;  // Model not found
    }

    @AuraEnabled
    public static String processWithEinstein(String modelId, String prompt, String context) {
        try {
            AI_Model_Configuration__mdt modelConfig = getModelConfiguration(modelId);
            system.debug('modelConfig::::'+modelConfig);
            System.debug('modelConfig != null:::::'+modelConfig != null);
            if (modelConfig != null) {
                system.debug('modelConfig.Model_Provider__c:::::'+modelConfig.Model_Provider__c);
                if (modelConfig.Model_Provider__c == 'einstein_intent') {
                    System.debug('inside if condition::::');
                    return processWithEinsteinNative(prompt, context, modelConfig);
                } else if (modelConfig.Model_Provider__c == 'OpenAI') {
                    throw new AIProcessingException('OpenAI provider not yet implemented.');
                    
                } else {
                    throw new AIProcessingException('Unsupported model provider: ' +
                                                    modelConfig.Model_Provider__c);
                }
                
            } else if (modelId.startsWith('einstein_')) {
                return processWithEinsteinNative(prompt, context, null);
            }
            
            throw new AIProcessingException('Model configuration not found for ID: ' + modelId);
            
        } catch (Exception e) {
            throw new AuraHandledException('AI processing failed: ' + e.getMessage());
        }
    }

   private static String processWithEinsteinNative(
    String prompt,
    String context,
    AI_Model_Configuration__mdt config
) {
    System.debug('prompt::::' + prompt);
    System.debug('context:::' + context);
    System.debug('config::::' + config);

    HttpRequest req = new HttpRequest();
    req.setEndpoint('callout:OpenAI/v1/chat/completions'); 
    req.setMethod('POST');
    req.setHeader('Content-Type', 'application/json');
    // ✅ No Authorization header (Named Credential handles it)

    Map<String, Object> body = new Map<String, Object>{
        'model' => 'gpt-4o-mini',   // ✅ HARDCODED VALID MODEL
        'messages' => new List<Object>{
            new Map<String, Object>{
                'role' => 'user',
                'content' => prompt + '\n\nContext:\n' + context
            }
        }
    };

    req.setBody(JSON.serialize(body));

    Http http = new Http();
    HttpResponse response = http.send(req);

    // ✅ Debug full response
    System.debug('STATUS CODE: ' + response.getStatusCode());
    System.debug('RAW RESPONSE: ' + response.getBody());

    if (response.getStatusCode() == 200) {
        Map<String, Object> result =
            (Map<String, Object>) JSON.deserializeUntyped(response.getBody());

        List<Object> choices = (List<Object>) result.get('choices');
        Map<String, Object> firstChoice =
            (Map<String, Object>) choices[0];
        Map<String, Object> message =
            (Map<String, Object>) firstChoice.get('message');

        return (String) message.get('content');
    }

    throw new AIProcessingException(
        'OpenAI API failed with status ' +
        response.getStatusCode() +
        ' | Response: ' + response.getBody()
    );
}

public with sharing class AutomatedDocumentGenerator {

    @InvocableMethod(label='Generate Document from Template' description='Generates a document for a given record using a specified template.')
    public static void generateDocument(List<Request> requests) {
        if (requests == null || requests.isEmpty()) {
            return;
        }

        Request req = requests[0]; // Process one request at a time for simplicity
        
        try {
            // 1. Fetch the template
            DocumentTemplate__c template = [
                SELECT Body__c, Name 
                FROM DocumentTemplate__c 
                WHERE Name = :req.templateName AND IsActive__c = true 
                LIMIT 1
            ];

            // 2. Fetch data from the source record (e.g., Opportunity)
            SObject sourceRecord = Database.query('SELECT Name, Amount, CloseDate FROM ' + req.recordId.getSObjectType().getDescribe().getName() + ' WHERE Id = :req.recordId');
            
            // 3. Assemble the document using your existing engine
            String mergeData = JSON.serialize(sourceRecord.getPopulatedFieldsAsMap());
            String assembledHtml = DocumentAssemblyEngine.assemble(template.Body__c, mergeData, null);

            // 4. Save the assembled document as a file
            ContentVersion cv = new ContentVersion();
            cv.Title = template.Name + ' - ' + sourceRecord.get('Name');
            cv.PathOnClient = cv.Title + '.html';
            cv.VersionData = Blob.valueOf(assembledHtml);
            cv.FirstPublishLocationId = req.recordId; // Link the file to the record
            insert cv;

        } catch (Exception e) {
            // In a real scenario, you would add more robust error handling,
            // like logging to a custom object or sending a notification.
            System.debug('Automated document generation failed: ' + e.getMessage());
        }
    }

    // Inner class to define the inputs for the invocable method
    public class Request {
        @InvocableVariable(required=true label='Source Record ID')
        public Id recordId;

        @InvocableVariable(required=true label='Template Name')
        public String templateName;
    }
}

     * @param region

     * @param aiModelIdentifier
     */
    @AuraEnabled
    public static String generateClause(
        String region,
        String role,
        String contractType,
        String aiModelIdentifier
    ) {
        try {
            if (String.isBlank(region) ||
                String.isBlank(role) ||
                String.isBlank(contractType)) {
                throw new AuraHandledException('All parameters (region, role, contractType) are required.');
            }

            String baseClause = getBaseClause(region, contractType);
            if (String.isBlank(baseClause)) {
                throw new AuraHandledException(
                    'No template found for Region: ' + region + ', Contract Type: ' + contractType
                );
            }

            String roleClause = ROLE_SPECIFIC_CLAUSES.get(role);
            if (roleClause == null) {
                roleClause =
                    '\n\nGENERAL PROVISIONS:\n'
                    + '- Standard terms and conditions apply';
            }

            String fullClause = baseClause + roleClause;

            if (String.isNotBlank(aiModelIdentifier)) {
                fullClause = enhanceWithAI(
                    fullClause,
                    region,
                    role,
                    contractType,
                    aiModelIdentifier
                );
            }

            logClauseGeneration(region, role, contractType, aiModelIdentifier, fullClause, 'SUCCESS');

            return fullClause;
        } catch (Exception e) {
            logClauseGeneration(region, role, contractType, aiModelIdentifier, e.getMessage(), 'ERROR');
            throw new AuraHandledException('Clause generation failed: ' + e.getMessage());
        }
    }

        String baseClause,

        String role,
        String contractType,

    ) {
        try {
            AIModelManager.AIModelConfig config = AIModelManager.getAIModelByIdentifier(aiModelIdentifier);

            if (config == null || !config.isActive) {
                return baseClause;
            }

            String enhanced = baseClause;

            enhanced += '\n\nThis agreement incorporates regulatory requirements as of '

        String region,
        String role,
        String contractType,
        Integer variationCount,
        String aiModelIdentifier
    ) {
        List<String> results = new List<String>();

        if (variationCount == null || variationCount <= 0) {
            variationCount = 3;
        }

        String baseClause = generateClause(region, role, contractType, aiModelIdentifier);
        results.add(baseClause);

        for (Integer i = 1; i < variationCount; i++) {
            results.add(generateClauseVariation(baseClause, i));
        }

        return results;
    }

    // =====================================================
    // INDIVIDUAL VARIATION GENERATOR
    // =====================================================
    private static String generateClauseVariation(String baseClause, Integer type) {
        String variation = baseClause;

        switch on type {
            when 1 {
                variation = variation.replace('you acknowledge', 'the party acknowledges');
                variation = variation.replace('You agree', 'The party agrees');
                variation += '\n\nFORMAL ADDENDUM: This agreement is the full understanding between parties.';
            }
            when 2 {
                variation +=
                    '\n\nDETAILED PROVISIONS: Includes performance metrics, compliance checkpoints, '
                    + 'and regular reviews.';
            }
            when else {
                variation +=
                    '\n\nIMPLEMENTATION NOTE: All provisions take effect immediately upon signature.';
            }
        }

        return variation;
    }

     * @param documentType The type of document (e.g., 'Employment', 'NDA').
     * @return Boolean - true if compliant, false if critical keywords are missing.
     */
    @AuraEnabled
    public static Boolean validateClause(String clauseText, String region, String documentType) {
        if (String.isBlank(clauseText)) return false;

        String contentLower = clauseText.toLowerCase();
        List<String> missingTerms = new List<String>();
        
        // Find rules based on region and document type
        Map<String, List<String>> regionRules = COMPLIANCE_RULES.get(region.toUpperCase());
        if (regionRules != null && regionRules.containsKey(documentType)) {
            List<String> requiredTerms = regionRules.get(documentType);
            
            for (String term : requiredTerms) {
                if (!contentLower.contains(term.toLowerCase())) {
                    missingTerms.add(term);
                }
            }
        } else {
            // If specific rules aren't found, treat as compliant, but log
            System.debug(LoggingLevel.WARN, 'No specific rules found for ' + region + '/' + documentType);
            return true;
        }

        if (missingTerms.isEmpty()) {
            // Log success implicitly or via audit if needed
            return true;
        } else {
            // Log violation and detail missing terms
            String violationDetails = 'Missing critical terms: ' + String.join(missingTerms, ', ');
            System.debug('COMPLIANCE VIOLATION: ' + violationDetails);
            // This flag can be passed to an AuditLogger if required by the full implementation spec
            return false;
        }
    }

            // Perform basic check using the new validateClause logic

                res.complianceScore = res.complianceScore - 20; // Deduct score for missing key terms
                res.violations.add('Missing critical regulatory terms for ' + jurisdiction + '/' + documentType);
            }

            // Example: HIPAA check based on content keywords

                res.violations.add('HIPAA: PHI identified without explicit encryption reference');

    //======================================================
    // MAIN ASSEMBLY METHOD – WITH AI MODEL INTEGRATION
    //======================================================
    @AuraEnabled
    public static String assemble(String templateContent, String mergeData, String aiModel) {
        try {
            // Parse merge data
            Map<String, Object> dataMap = (Map<String, Object>) JSON.deserializeUntyped(mergeData);

            // Standard merge field substitution
            String assembledContent = performStandardSubstitution(templateContent, dataMap);

            // Apply optional AI enhancement
            if (String.isNotBlank(aiModel)) {
                assembledContent = enhanceWithAI(assembledContent, dataMap, aiModel);
            }

            return assembledContent;

        } catch (Exception e) {
            throw new AuraHandledException('Document assembly failed: ' + e.getMessage());
        }
    }


    //======================================================
    // STANDARD MERGE FIELD SUBSTITUTION
    //======================================================
    private static String performStandardSubstitution(String template, Map<String, Object> dataMap) {
        String result = template;

        for (String key : dataMap.keySet()) {
            Object value = dataMap.get(key);

            // Nested object (e.g., contact.Name)
            if (value instanceof Map<String, Object>) {
                Map<String, Object> nestedMap = (Map<String, Object>) value;

                for (String nestedKey : nestedMap.keySet()) {
                    String mergeField = '{{' + key + '.' + nestedKey + '}}';
                    String altField = '{' + key + '.' + nestedKey + '}';
                    String replacement = String.valueOf(nestedMap.get(nestedKey));

                    result = result.replace(mergeField, replacement);
                    result = result.replace(altField, replacement);
                }
            }

            // Simple field (e.g., {{documentType}})
            else {
                String mergeField = '{{' + key + '}}';
                String altField = '{' + key + '}';
                String replacement = String.valueOf(value);

                result = result.replace(mergeField, replacement);
                result = result.replace(altField, replacement);
            }
        }

        return result;
    }

    //======================================================

     * @description Executes a list of rules against a set of JSON data.
     * @param rules The list of rules to execute.
     * @param jsonData The data to validate.
     * @return DocumentGenerationController.ComplianceResult The outcome of the validation.
     */
    public static DocumentGenerationController.ComplianceResult execute(List<ComplianceRule__mdt> rules, String jsonData) {
        List<String> errorMessages = new List<String>();
        Map<String, Object> data = (Map<String, Object>) JSON.deserializeUntyped(jsonData);

        for (ComplianceRule__mdt rule : rules) {
            // NOTE: This is a simplified rule evaluator for demonstration.
            // Example RuleLogic__c: "opportunity.Amount!= null"
            String fieldPath = rule.RuleLogic__c.split(' ')[0]; // e.g., "opportunity.Amount"
            Object value = getValueFromPath(data, fieldPath);

            if (value == null || (value instanceof String && String.isBlank((String)value))) {
                errorMessages.add(rule.ErrorMessage__c);
            }
        }
        
        Boolean isCompliant = errorMessages.isEmpty();
        return new DocumentGenerationController.ComplianceResult(isCompliant, errorMessages);
    }

    /**
     * @description Helper to navigate a JSON map using a dot-notation path.
     */
    private static Object getValueFromPath(Map<String, Object> data, String path) {
        List<String> keys = path.split('\\.');
        Object currentObject = data;

        for (String key : keys) {
            if (currentObject instanceof Map<String, Object>) {
                currentObject = ((Map<String, Object>)currentObject).get(key);
            } else {
                return null; // Path is invalid
            }
        }
        return currentObject;
    }

    @track currentStep = 'step1';
    @track isLoading = false;
    @track selectedTemplateId = '';
    @track jsonData = '{\n "contact": {\n "Name": "John Doe",\n "IsUSCitizen": true\n }\n}';
    @track templateOptions = [];

    @track validationErrors = [];

    @track selectedAIModelId = '';
    @track aiPrompt = 'Suggest a standard confidentiality clause.';

            this.showToast('Error', 'Failed to load templates.', 'error');
        }
    }

    // Wire to get available AI models
    @wire(getAvailableAIModels)
    wiredAIModels({ error, data }) {
        if (data) {
            this.aiModelOptions = data.map(model => ({
                label: `${model.modelName} (${model.provider})`,
                value: model.modelIdentifier
            }));
        } else if (error) {
            this.showToast('Error', 'Failed to load AI models.', 'error');
        }
    }

    // --- Event Handlers for Step 1 ---
    handleTemplateChange(event) {
        this.selectedTemplateId = event.detail.value;
    }

    handleJsonChange(event) {
        this.jsonData = event.target.value;
    }

    // NEW: Handler for the AI feature toggle
    handleAIToggle(event) {
        this.aiFeaturesEnabled = event.detail.checked;
        // If user turns AI off, clear any previously generated clause
        if (!this.aiFeaturesEnabled) {
            this.generatedClause = '';
        }
    }

    handleAIModelChange(event) {
        this.selectedAIModelId = event.detail.value;
    }

    handleAIPromptChange(event) {
        this.aiPrompt = event.target.value;
    }

    async handleSuggestClauseClick() {
        if (!this.selectedAIModelId) {
            this.showToast('Warning', 'Please select an AI model first.', 'warning');
            return;
        }
        this.isGeneratingClause = true;
        this.generatedClause = '';
        try {
            const result = await generateClauseWithAI({
                modelId: this.selectedAIModelId,
                prompt: this.aiPrompt,
                jsonContext: this.jsonData
            });
            this.generatedClause = result;
        } catch (error) {
            this.showToast('Error', error.body?.message || 'Failed to generate AI clause.', 'error');
        } finally {
            this.isGeneratingClause = false;
        }
    }

    // --- Navigation and Core Logic ---
    handleBackClick() {
        this.currentStep = 'step1';
        this.previewHtml = '';
        this.validationErrors = [];
    }

    async handlePreviewClick() {
        if (!this.selectedTemplateId || !this.jsonData) {
            this.showToast('Warning', 'Please select a template and provide JSON data.', 'warning');
            return;
        }
        this.isLoading = true;
        try {
            // Pass the generated clause only if AI features are enabled
            const clauseToSend = this.aiFeaturesEnabled ? this.generatedClause : '';
            const result = await previewDocument({
                templateId: this.selectedTemplateId,
                jsonData: this.jsonData,
                aiClauseText: clauseToSend
            });
            if (result.success) {
                this.previewHtml = result.renderedContent;
                this.validationErrors = [];
                this.currentStep = 'step2';
            } else {
                this.validationErrors = result.validationErrors;
                this.previewHtml = '';
                this.currentStep = 'step2';
            }
        } catch (error) {
            this.validationErrors = [error.body?.message || 'An unknown server error occurred.'];
            this.previewHtml = '';
        } finally {
            this.isLoading = false;
        }
    }

    async handleGenerateClick() {
        this.isLoading = true;
        try {
            // Pass the generated clause only if AI features are enabled
            const clauseToSend = this.aiFeaturesEnabled ? this.generatedClause : '';
            const result = await generateFinalDocument({
                templateId: this.selectedTemplateId,
                jsonData: this.jsonData,
                aiClauseText: clauseToSend,
                selectedAIModel: this.selectedAIModelId
            });
            if (result.success && result.recordId) {
                this.showToast('Success', 'Document generated successfully!', 'success');
                this.generationComplete = true;
                this.newRecordId = result.recordId;
                this.newContentDocumentId = result.contentDocumentId;
            } else {
                this.showToast('Error', result.errorMessage, 'error');
            }
        } catch (error) {
            this.showToast('Error', error.body?.message || 'An unknown error occurred.', 'error');
        } finally {
            this.isLoading = false;
        }
    }

    // --- Success Screen Handlers ---
    handleViewDocument() {
        this[NavigationMixin.Navigate]({
            type: 'standard__namedPage',
            attributes: {
                pageName: 'filePreview'
            },
            state: {
                recordIds: this.newContentDocumentId
            }
        });
    }

    handleDownloadDocument() {
        const downloadUrl = `/sfc/servlet.shepherd/document/download/${this.newContentDocumentId}`;
        window.open(downloadUrl, '_blank');
    }

    handleGenerateNew() {
        this.resetComponent();
        this.showToast('Ready', 'You can now generate a new document.', 'success');
    }

    // --- Utility Methods ---
    resetComponent() {
        this.currentStep = 'step1';
        this.isLoading = false;
        this.selectedTemplateId = '';
        this.previewHtml = '';
        this.validationErrors = [];
        this.generationComplete = false;
        this.newRecordId = '';
        this.newContentDocumentId = '';

        // Reset AI fields
        this.aiFeaturesEnabled = false; // Turn toggle off
        this.selectedAIModelId = '';
        this.generatedClause = '';
        this.aiPrompt = 'Suggest a standard confidentiality clause.';
    }

    showToast(title, message, variant) {
        this.dispatchEvent(
            new ShowToastEvent({
                title,
                message,
                variant
            })
        );
    }
}

        }
    }

        this.isEditing = false;
        this.selectedTemplate = {};
        const datatable = this.template.querySelector('lightning-datatable');
        if (datatable) {
            datatable.selectedRows = [];
        }
    }

    showToast(title, message, variant) {
        this.dispatchEvent(new ShowToastEvent({ title, message, variant }));
    }
